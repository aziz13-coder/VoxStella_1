> "All the Planets act by direction as well as by transit **in accordance with their own radical determination** and their own nature"

```python
def calculate_transit_quality(transit, natal_chart, determinations, 
                              solar_rev, lunar_rev):
    """
    Returns: quality_score (-10 to +10), quality_label, confidence
    """
    
    # STEP 1: Get Radical Determination Score
    transiting_planet = transit.planet
    target_life_area = identify_life_area(transit.target, natal_chart)
    
    determination_score = get_determination_score(
        transiting_planet, 
        target_life_area, 
        determinations
    )
    # Range: -1.0 (determined against) to +1.0 (determined for)
    
    # STEP 2: Get Planetary Base Nature
    base_nature = PLANET_NATURES[transiting_planet]
    # Jupiter/Venus: +1, Sun/Moon: 0, Mercury: 0, Mars: -1, Saturn: -2
    
    # STEP 3: Aspect Modification
    aspect_modifier = ASPECT_MODIFIERS[transit.aspect_type]
    # Conjunction/Opposition/Square: 1.0 (full strength)
    # Trine/Sextile: 0.8 (strong but easier)
    # Minor aspects: 0.5
    
    # STEP 4: Dignity Modification
    dignity_mult = calculate_dignity_multiplier(
        transiting_planet, 
        natal_chart
    )
    # Range: 0.7 (debilitated) to 1.2 (dignified)
    
    # STEP 5: Angular/House Modification
    target_importance = calculate_target_importance(transit.target)
    # Angular: 1.3, Succedent: 1.1, Cadent: 0.9
    
    # STEP 6: Revolution Condition
    revolution_modifier = calculate_revolution_condition(
        transiting_planet,
        solar_rev,
        lunar_rev
    )
    # Range: 0.7 (badly placed) to 1.3 (well placed)
    
    # STEP 7: Concordance Multiplier
    concordance_mult = calculate_concordance_multiplier(
        transit.concordance.overall_concordance
    )
    # Range: 0.3 (no concordance) to 2.0 (high concordance)
    
    # CALCULATION
    # Primary component: DETERMINATION (this is Morin's key insight)
    primary_quality = determination_score * 5  # -5 to +5
    
    # Secondary component: Natural benefic/malefic
    secondary_quality = base_nature * 2  # -4 to +2
    
    # Combine (determination is MORE important)
    base_quality = (primary_quality * 0.7) + (secondary_quality * 0.3)
    
    # Apply modifiers
    modified_quality = (
        base_quality * 
        aspect_modifier * 
        dignity_mult * 
        target_importance * 
        revolution_modifier *
        concordance_mult
    )
    
    # Clamp to -10 to +10
    quality_score = max(-10, min(10, modified_quality))
    
    # Determine label
    if quality_score > 7:
        quality_label = "very_benefic"
    elif quality_score > 4:
        quality_label = "benefic"
    elif quality_score > 1:
        quality_label = "moderately_benefic"
    elif quality_score > -1:
        quality_label = "neutral"
    elif quality_score > -4:
        quality_label = "moderately_malefic"
    elif quality_score > -7:
        quality_label = "malefic"
    else:
        quality_label = "very_malefic"
    
    # Confidence based on concordance
    confidence = min(0.95, concordance_mult / 2.0)
    
    return quality_score, quality_label, confidence
```

class TransitQualityAnalyzer:
    
    def analyze_transit_quality(self, transit, natal_chart, 
                                 determinations, revolutions):
        """
        Main quality determination function
        """
        
        # 1. GET RADICAL DETERMINATION (MOST IMPORTANT)
        det_score = self.get_determination_match(
            transit.planet,
            transit.target,
            determinations

        base_nature = PLANET_BASE_NATURES[transit.planet]

        )
        
        # 5. DIGNITY/CONDITION
        dignity = self.calculate_dignity(transit.planet, natal_chart)
        revolution_state = self.check_revolution_state(
            transit.planet, 
            revolutions
        )

        concordance = transit.concordance.overall_concordance
        
        # 7. COMBINE WITH MORIN'S WEIGHTS
        quality_score = self.combine_factors(
            determination_score = det_score,     # 50% weight
            base_nature = base_nature,           # 20% weight
            aspect = aspect_mod,                 # 15% weight
            dignity = dignity,                   # 10% weight
            revolution_state = revolution_state  # 5% weight
        ) * self.concordance_multiplier(concordance)
        
        # 8. CHECK SPECIAL CASES (Morin's Laws)
        quality_score = self.apply_special_rules(
            quality_score, 
            transit, 
            natal_chart
        )
        
        # 9. CONVERT TO LABEL
        quality_label = self.score_to_label(quality_score)
        
        # 10. CALCULATE CONFIDENCE
        confidence = self.calculate_confidence(
            concordance,
            determination_certainty,
            data_quality
        )
        
        return {
            'score': quality_score,
            'label': quality_label,
            'confidence': confidence,
            'primary_factor': 'determination',
            'determination_score': det_score,
            'breakdown': self.get_breakdown()
        }
    
    
    def get_determination_match(self, planet, target, determinations):
        """
        This is THE KEY FUNCTION
        Returns -1.0 to +1.0
        """
        target_life_area = self.identify_life_area(target)
        
        planet_det = determinations[planet]
        
        # Get determination score for this life area
        score = planet_det.determination_scores[target_life_area]
        
        # Check for contrary determination
        # (e.g., planet determined to death hitting ASC)
        if self.is_contrary_determination(planet_det, target_life_area):
            score = -abs(score)  # Make negative
        
        return score
    
    
    def get_aspect_modifier(self, aspect_type, determination_score):
        """
        Aspect modifier DEPENDS on determination
        Good determination: even "hard" aspects help (but with friction)
        Bad determination: even "soft" aspects harm (but subtly)
        """
        
        base_aspect_strength = {
            'conjunction': 1.0,
            'opposition': 1.0,
            'square': 0.9,
            'trine': 0.8,
            'sextile': 0.7,
            'quincunx': 0.5
        }[aspect_type]
        
        # If determination is positive, keep aspect strength as-is
        # If determination is negative, aspect amplifies the negativity
        
        if determination_score > 0:
            # Benefic determination
            if aspect_type in ['conjunction', 'trine', 'sextile']:
                return base_aspect_strength * 1.2  # Extra boost
            else:
                return base_aspect_strength * 1.0  # Full strength but tense
        else:
            # Malefic determination
            if aspect_type in ['conjunction', 'opposition', 'square']:
                return base_aspect_strength * 1.3  # Amplified harm
            else:
                return base_aspect_strength * 0.9  # Subtle harm
    
    
    def apply_special_rules(self, base_score, transit, natal_chart):
        """
        Apply Morin's specific laws that override normal calculations
        """

        # Law 7: Return to radical place
        if self.is_return_to_radical(transit):
            score *= 1.5  # Amplify whatever the base score is
        
        # Law 10: Malefics to ASC
        if self.is_malefic_to_asc(transit, natal_chart):
            if self.malefic_rules_8th_or_12th(transit.planet, natal_chart):
                score = min(score, -8)  # Force very malefic
        
        # Law 13: Lights conjunct malefics
        if self.is_lights_conjunct_malefic(transit, natal_chart):
            if self.has_death_direction(transit):
                score = -10  # Maximum danger
        
        return score
