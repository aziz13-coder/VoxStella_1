def calculate_determination_strength(planet, life_area, natal_chart):
    """
    Calculate how strongly a planet is determined to a specific life area
    Returns: 0.0 to 1.0 score
    """
    score = 0.0
    
    # 1. Natural Signification Match (weight: 0.3)
    if life_area in PLANET_KEYWORDS[planet]['natural_significations']:
        score += 0.3
    
    # 2. House Position Match (weight: 0.4)
    planet_house = get_house_position(planet, natal_chart)
    if life_area in HOUSE_KEYWORDS[planet_house]['life_areas']:
        score += 0.4
    
    # 3. Rulership Match (weight: 0.5 per rulership)
    for house in get_ruled_houses(planet, natal_chart):
        if life_area in HOUSE_KEYWORDS[house]['life_areas']:
            score += 0.5 * get_rulership_strength_multiplier(planet, house)
    
    # 4. Aspect-based Determination (weight: 0.2 per relevant aspect)
    for aspect in get_aspects(planet, natal_chart):
        aspected_planet = aspect.planet
        aspected_house = get_house_position(aspected_planet, natal_chart)
        if life_area in HOUSE_KEYWORDS[aspected_house]['life_areas']:
            aspect_strength = ASPECT_KEYWORDS[aspect.type]['strength']
            score += 0.2 * aspect_strength
    
    # Normalize to 0-1 range
    max_possible = 2.5  # Theoretical maximum
    normalized_score = min(1.0, score / max_possible)
    
    return normalized_score
```

def match_event_type(determination_scores, quality, planets_involved):
    """
    Determine the most likely event type based on determination and quality
    Returns: event_type, confidence
    """
    event_matches = []
    
    # Get top life areas from determination scores
    top_areas = sorted(
        determination_scores.items(), 
        key=lambda x: x[1], 
        reverse=True
    )[:3]
    
    # For each potential event type
    for event_type, event_data in EVENT_TYPE_KEYWORDS.items():
        match_score = 0.0
        
        # Check if life areas match
        for life_area, strength in top_areas:
            if life_area in event_data['life_areas']:
                match_score += strength * 0.4
        
        # Check if planets match
        for planet in planets_involved:
            if planet in event_data['planets']:
                match_score += 0.3
        
        # Check if quality matches
        if quality == event_data['quality']:
            match_score += 0.3
        elif quality in ['benefic', 'malefic'] and event_data['quality'] == 'mixed':
            match_score += 0.15
        
        event_matches.append((event_type, match_score))
    
    # Get best match
    best_match = max(event_matches, key=lambda x: x[1])
    
    return best_match[0], best_match[1]
```

```python
def calculate_signification_match(significations_1, significations_2):
    """
    Calculate how well two sets of significations match
    Returns: 0.0 to 1.0 score
    """
    # Direct matches
    direct_matches = len(
        set(significations_1) & set(significations_2)
    )
    
    # Related matches (from semantic network)
    related_matches = 0
    for sig1 in significations_1:
        for sig2 in significations_2:
            if are_related_significations(sig1, sig2):
                related_matches += 0.5
    
    # Opposite matches (negative correlation)
    opposite_matches = 0
    for sig1 in significations_1:
        for sig2 in significations_2:
            if are_opposite_significations(sig1, sig2):
                opposite_matches -= 0.3
    
    total_score = direct_matches + related_matches + opposite_matches
    
    # Normalize
    max_possible = len(significations_1)
    normalized = min(1.0, max(0.0, total_score / max_possible))
    
    return normalized


def are_related_significations(sig1, sig2):
    """Check semantic relationship network"""
    for group in SEMANTIC_RELATIONSHIPS['Related_Significations'].values():
        if sig1 in group and sig2 in group:
            return True
    return False


def are_opposite_significations(sig1, sig2):
    """Check opposition network"""
    for opposite_pair in SEMANTIC_RELATIONSHIPS['Opposite_Significations'].values():
        if (sig1 in opposite_pair['positive'] and 
            sig2 in opposite_pair['negative']) or \
           (sig1 in opposite_pair['negative'] and 
            sig2 in opposite_pair['positive']):
            return True
    return False
```
